cmake_minimum_required(VERSION 3.13)
project(ChibiAviDecoder LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} /MP /EHsc /DNOMINMAX")
set(GODOT_CPP_WARNING_AS_ERROR OFF CACHE BOOL "" FORCE)
add_subdirectory(godot-cpp-4.1)
file(GLOB SRC "src/*.cpp")
add_library(ChibiAviDecoder SHARED ${SRC})
target_include_directories(ChibiAviDecoder PRIVATE FFmpeg-n6.1.2 godot-cpp-4.1/gdextension)
target_link_libraries(ChibiAviDecoder PRIVATE godot::cpp "${CMAKE_CURRENT_SOURCE_DIR}/lib/avcodec.lib" "${CMAKE_CURRENT_SOURCE_DIR}/lib/avformat.lib" "${CMAKE_CURRENT_SOURCE_DIR}/lib/avutil.lib" "${CMAKE_CURRENT_SOURCE_DIR}/lib/swresample.lib" "${CMAKE_CURRENT_SOURCE_DIR}/lib/swscale.lib")
set_target_properties(ChibiAviDecoder PROPERTIES OUTPUT_NAME "FFmpegPlayer")
