cmake_minimum_required(VERSION 3.15)
project(ChibiSimulationMath LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Python3 REQUIRED)
set(BINDINGS "${CMAKE_CURRENT_SOURCE_DIR}/../../../_work/sim-math/godot-cpp")
set(GENERATED "${CMAKE_CURRENT_BINARY_DIR}/bindings")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/profile.json" "{\"enabled_classes\":[\"RefCounted\",\"AStar2D\"]}")
if(NOT EXISTS "${GENERATED}/gen/include/godot_cpp/classes/object.hpp")
execute_process(COMMAND ${Python3_EXECUTABLE} -c "import binding_generator; binding_generator.generate_bindings('gdextension/extension_api.json',False,'64','single',r'${GENERATED}')" WORKING_DIRECTORY ${BINDINGS} RESULT_VARIABLE GENERATED_RESULT)
if(NOT GENERATED_RESULT EQUAL 0)
 message(FATAL_ERROR "Binding generation failed")
endif()
endif()
execute_process(COMMAND ${Python3_EXECUTABLE} -c "import binding_generator; print(';'.join(binding_generator.get_file_list('gdextension/extension_api.json',r'${GENERATED}',sources=True,profile_filepath=r'${CMAKE_CURRENT_BINARY_DIR}/profile.json')))" WORKING_DIRECTORY ${BINDINGS} OUTPUT_VARIABLE GEN_SOURCES OUTPUT_STRIP_TRAILING_WHITESPACE)
string(REGEX REPLACE "Using feature build profile:[^\n]*\n" "" GEN_SOURCES "${GEN_SOURCES}")
file(GLOB_RECURSE CORE "${BINDINGS}/src/*.cpp")
add_library(bindings STATIC ${CORE} ${GEN_SOURCES})
target_include_directories(bindings PUBLIC ${BINDINGS}/include ${BINDINGS}/gdextension ${GENERATED}/gen/include)
target_compile_definitions(bindings PUBLIC TYPED_METHOD_BIND NOMINMAX)
target_compile_options(bindings PRIVATE /utf-8)
add_library(SimMath SHARED sim_math.cpp)
target_link_libraries(SimMath PRIVATE bindings)
target_compile_definitions(SimMath PRIVATE SIM_MATH_SIGNATURE="${SIM_SIGNATURE}")
target_compile_options(SimMath PRIVATE /fp:precise /utf-8)
set_target_properties(SimMath PROPERTIES OUTPUT_NAME ChibiSimulationMath)
set_target_properties(bindings SimMath PROPERTIES INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
